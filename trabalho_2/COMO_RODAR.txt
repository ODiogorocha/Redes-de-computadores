# Trabalho 2 – Telemetria P4  |  Guia de Execução
UFSM – Departamento de Computação Aplicada

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
## Estrutura de arquivos

  trabalho2/
  ├── telemetria_trabalho2.p4   – programa P4 (switch)
  ├── topo_trabalho2.json       – topologia Mininet
  ├── s1-runtime.json           – regras via P4Runtime
  ├── s1-commands.txt           – regras via simple_switch_CLI
  ├── dashboard.py              – coletor + dashboard Tkinter (h3)
  ├── grupo.txt                 – integrantes do grupo
  └── relatorio_trabalho2.md    – relatório

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
## Pré-requisitos

  sudo apt install p4lang-p4c behavioral-model mininet python3-tk

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
## PASSO 1 – Compilar o programa P4

  mkdir -p build

  p4c --target bmv2 \
      --arch v1model \
      --p4runtime-files build/telemetria_trabalho2.p4.p4info.txt \
      -o build/ \
      telemetria_trabalho2.p4

  # Resultado esperado:
  #   build/telemetria_trabalho2.json
  #   build/telemetria_trabalho2.p4.p4info.txt

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
## PASSO 2 – Iniciar o Mininet com o switch P4

  sudo mn \
    --custom topo_trabalho2.json \
    --topo trabalho2 \
    --switch simple_switch_grpc \
    --controller none

  # Ou, usando o script padrão do p4-utils:
  sudo python3 /usr/local/bin/p4run --config topo_trabalho2.json

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
## PASSO 3 – Instalar as regras no switch

  ### Opção A – simple_switch_CLI (mais simples)

    simple_switch_CLI --thrift-port 9090 < s1-commands.txt

  ### Opção B – P4Runtime

    python3 /path/to/runtime_CLI.py \
      --p4info build/telemetria_trabalho2.p4.p4info.txt \
      --bmv2-json build/telemetria_trabalho2.json \
      --runtime-json s1-runtime.json

  # As regras definem:
  #   - encaminhamento IPv4 para h1, h2, h3
  #   - sessão de clone 99 → porta 3 (h3)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
## PASSO 4 – Iniciar o Dashboard em h3

  Na CLI do Mininet:

    mininet> xterm h3

  Na janela que abrir (xterm de h3):

    sudo python3 dashboard.py --iface h3-eth0

  Ou diretamente da CLI do Mininet em background:

    mininet> h3 sudo python3 dashboard.py --iface h3-eth0 &

  A janela Tkinter abrirá mostrando:
    • Pacotes / janela
    • Bytes / janela
    • TTL mínimo
    • Tamanho médio dos pacotes
    • Protocolo dominante (TCP / UDP / ICMP)
    • Log de todas as janelas com timestamp

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
## PASSO 5 – Gerar tráfego entre os hosts

  Na CLI do Mininet:

    # Ping (ICMP) – gera janelas de telemetria rapidamente
    mininet> h1 ping -c 50 10.0.2.2

    # Tráfego UDP com iperf
    mininet> h2 iperf -u -s &
    mininet> h1 iperf -u -c 10.0.2.2 -t 30 -b 1M

    # Tráfego TCP com iperf
    mininet> h2 iperf -s &
    mininet> h1 iperf -c 10.0.2.2 -t 30

  A cada 10 pacotes observados pelo switch, um relatório é
  enviado para h3 e aparece automaticamente no dashboard.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
## Verificação rápida

  # Confirmar que h3 recebe os pacotes de telemetria:
  mininet> h3 tcpdump -i h3-eth0 -n -e ether proto 0x9999

  # Deve aparecer pacotes a cada 10 pkts de h1→h2 ou h2→h1

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
## Resumo das métricas exportadas

  1. pkt_count      – pacotes na janela (obrigatório)
  2. byte_count     – bytes na janela   (obrigatório)
  3. min_ttl        – TTL mínimo observado (adicional)
  4. dominant_proto – protocolo mais frequente (adicional)

  Janela de observação: N = 10 pacotes (constante WINDOW_SIZE no .p4)
